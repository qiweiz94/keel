-- probe plugin: /tmp/keel-oc-probe-1786467965/work/.opencode/plugins/probe-logger.js --
import fs from "node:fs"
const LOG = "/tmp/keel-oc-probe-log.jsonl"
function log(obj) {
  try { fs.appendFileSync(LOG, JSON.stringify(obj) + "\n") } catch (e) {}
}
export default {
  id: "keel-probe-logger",
  server: async (pluginInput) => {
    log({ hook: "server-init", directory: pluginInput?.directory })
    return {
      event: async ({ event }) => {
        // Only log a truncated shape for text-heavy events to keep the log readable.
        log({ hook: "event", type: event?.type, properties: event?.properties })
      },
      "experimental.text.complete": async (input, output) => {
        log({ hook: "experimental.text.complete", input, textLen: output?.text?.length, textSample: (output?.text || "").slice(0, 200) })
      },
      "tool.execute.before": async (input, output) => {
        log({ hook: "tool.execute.before", tool: input?.tool, sessionID: input?.sessionID })
      },
      "tool.execute.after": async (input, output) => {
        log({ hook: "tool.execute.after", tool: input?.tool, sessionID: input?.sessionID })
      },
      "chat.message": async (input, output) => {
        log({ hook: "chat.message", input })
      },
    }
  },
}
